// ====================================================== // KPL OFFICIAL - REGISTRATION.JS // FULL PLAYER DETAILS VERSION // ====================================================== import { auth, db, storage } from "./firebase.js"; import { onAuthStateChanged } from "https://www.gstatic.com/firebasejs/12.1.0/firebase-auth.js"; import { doc, getDoc, addDoc, collection, serverTimestamp } from "https://www.gstatic.com/firebasejs/12.1.0/firebase-firestore.js"; import { ref, uploadBytes, getDownloadURL } from "https://www.gstatic.com/firebasejs/12.1.0/firebase-storage.js"; // ====================================================== // ELEMENTS // ====================================================== const form = document.getElementById("registrationForm"); const tournamentId = new URLSearchParams(window.location.search).get("tournament"); // ====================================================== // GLOBAL // ====================================================== let currentUser = null; let tournamentData = null; let minPlayers = 1; let maxPlayers = 20; // ====================================================== // SAFE // ====================================================== function safe(value) { if ( value === null || value === undefined ) { return ""; } return String(value) .replaceAll("&", "&") .replaceAll("<", "<") .replaceAll(">", ">") .replaceAll('"', """) .replaceAll("'", "'"); } // ====================================================== // AUTH // ====================================================== onAuthStateChanged( auth, async (user) => { if (!user) { alert( "Please login with Google first." ); window.location.href = "index.html"; return; } currentUser = user; const email = document.getElementById("userEmail"); const userName = document.getElementById("loggedUserName"); if (email) { email.textContent = user.email || ""; } if (userName) { userName.textContent = user.displayName || "User"; } await loadTournament(); } ); // ====================================================== // LOAD TOURNAMENT // ====================================================== async function loadTournament() { if (!tournamentId) { alert( "Tournament ID missing." ); return; } try { const tournamentRef = doc( db, "tournaments", tournamentId ); const snap = await getDoc( tournamentRef ); if (!snap.exists()) { alert( "Tournament not found." ); return; } tournamentData = snap.data(); // ---------------------------------------------- // BASIC DETAILS // ---------------------------------------------- const tournamentName = getField( tournamentData, [ "name", "tournamentName", "title" ], "KPL Tournament" ); const sport = getField( tournamentData, [ "sport", "category", "type" ], "Football" ); const entryFee = getField( tournamentData, [ "entryFee", "entry_fee", "fee" ], "" ); const prizePool = getField( tournamentData, [ "prizePool", "prize_pool", "prize" ], "" ); // ---------------------------------------------- // PLAYER LIMIT // ---------------------------------------------- minPlayers = Number( getField( tournamentData, [ "minPlayers", "minimumPlayers", "minimum_players", "minPlayer" ], 1 ) ); maxPlayers = Number( getField( tournamentData, [ "maxPlayers", "maximumPlayers", "maximum_players", "maxPlayer" ], 20 ) ); if ( !Number.isFinite(minPlayers) || minPlayers < 1 ) { minPlayers = 1; } if ( !Number.isFinite(maxPlayers) || maxPlayers < minPlayers ) { maxPlayers = 20; } // ---------------------------------------------- // DISPLAY // ---------------------------------------------- setText( "tournamentName", tournamentName ); setText( "tournamentSport", sport ); setText( "tournamentEntryFee", entryFee ? "₹" + entryFee : "TBA" ); setText( "tournamentPrizePool", prizePool ? "₹" + prizePool : "TBA" ); setText( "playerLimit", `${minPlayers} - ${maxPlayers} Players` ); // ---------------------------------------------- // CREATE PLAYERS // ---------------------------------------------- createPlayerFields(); console.log( "Tournament loaded:", tournamentData ); } catch (error) { console.error( "Tournament loading error:", error ); alert( "Tournament loading failed: " + error.message ); } } // ====================================================== // FIELD HELPER // ====================================================== function getField( data, fields, fallback = "" ) { for ( const field of fields ) { if ( data[field] !== undefined && data[field] !== null && data[field] !== "" ) { return data[field]; } } return fallback; } // ====================================================== // TEXT HELPER // ====================================================== function setText( id, value ) { const element = document.getElementById(id); if (element) { element.textContent = value; } } // ====================================================== // CREATE PLAYER FIELDS // ====================================================== function createPlayerFields() { const container = document.getElementById( "playersContainer" ); if (!container) { console.error( "playersContainer not found." ); return; } container.innerHTML = ""; for ( let i = 1; i <= maxPlayers; i++ ) { const required = i <= minPlayers; const player = document.createElement( "div" ); player.className = "player-card"; player.dataset.player = i; player.innerHTML = `

Player ${i}

${ required ? ` REQUIRED ` : ` OPTIONAL ` }
JPG, PNG or WEBP
JPG, PNG or WEBP
`; container.appendChild( player ); } // ---------------------------------------------- // AUTO AGE FROM DOB // ---------------------------------------------- container .querySelectorAll( ".player-dob" ) .forEach( (dobInput) => { dobInput.addEventListener( "change", () => { const player = dobInput.dataset.player; const ageInput = container.querySelector( `.player-age[data-player="${player}"]` ); if (!ageInput) return; const age = calculateAge( dobInput.value ); if (age !== null) { ageInput.value = age; } } ); } ); } // ====================================================== // CALCULATE AGE // ====================================================== function calculateAge( dob ) { if (!dob) return null; const birth = new Date(dob); if ( isNaN( birth.getTime() ) ) { return null; } const today = new Date(); let age = today.getFullYear() - birth.getFullYear(); const month = today.getMonth() - birth.getMonth(); if ( month < 0 || ( month === 0 && today.getDate() < birth.getDate() ) ) { age--; } return age; } // ====================================================== // VALIDATE PLAYER // ====================================================== function validatePlayer( player, index ) { const name = player.name.trim(); const mobile = player.mobile.trim(); const dob = player.dob; const age = Number(player.age); const role = player.role; const aadhaar = player.aadhaar.trim(); if (!name) { throw new Error( `Player ${index}: Name is required.` ); } if ( !/^[0-9]{10}$/.test( mobile ) ) { throw new Error( `Player ${index}: Enter valid 10 digit mobile number.` ); } if (!dob) { throw new Error( `Player ${index}: Date of Birth is required.` ); } if ( !Number.isFinite(age) || age < 1 || age > 100 ) { throw new Error( `Player ${index}: Enter valid age.` ); } if (!role) { throw new Error( `Player ${index}: Select player role.` ); } if ( !/^[0-9]{12}$/.test( aadhaar ) ) { throw new Error( `Player ${index}: Aadhaar number must be 12 digits.` ); } if (!player.frontFile) { throw new Error( `Player ${index}: Aadhaar front photo is required.` ); } if (!player.backFile) { throw new Error( `Player ${index}: Aadhaar back photo is required.` ); } return true; } // ====================================================== // GET PLAYERS // ====================================================== function getPlayers() { const players = []; for ( let i = 1; i <= maxPlayers; i++ ) { const nameInput = document.querySelector( `.player-name[data-player="${i}"]` ); const mobileInput = document.querySelector( `.player-mobile[data-player="${i}"]` ); const dobInput = document.querySelector( `.player-dob[data-player="${i}"]` ); const ageInput = document.querySelector( `.player-age[data-player="${i}"]` ); const roleInput = document.querySelector( `.player-role[data-player="${i}"]` ); const aadhaarInput = document.querySelector( `.player-aadhaar[data-player="${i}"]` ); const frontInput = document.querySelector( `.player-aadhaar-front[data-player="${i}"]` ); const backInput = document.querySelector( `.player-aadhaar-back[data-player="${i}"]` ); if (!nameInput) continue; const name = nameInput.value.trim(); const mobile = mobileInput?.value.trim() || ""; const dob = dobInput?.value || ""; const age = ageInput?.value || ""; const role = roleInput?.value || ""; const aadhaar = aadhaarInput?.value.trim() || ""; const frontFile = frontInput?.files?.[0] || null; const backFile = backInput?.files?.[0] || null; const isEmpty = !name && !mobile && !dob && !age && !role && !aadhaar && !frontFile && !backFile; // Optional empty player if ( i > minPlayers && isEmpty ) { continue; } const player = { index: i, name, mobile, phone: mobile, mobileNumber: mobile, dob, dateOfBirth: dob, age: Number(age), role, aadhaarNumber: aadhaar, aadhaar: aadhaar, frontFile, backFile }; validatePlayer( player, i ); players.push( player ); } if ( players.length < minPlayers ) { throw new Error( `Minimum ${minPlayers} players are required.` ); } if ( players.length > maxPlayers ) { throw new Error( `Maximum ${maxPlayers} players are allowed.` ); } return players; } // ====================================================== // UPLOAD IMAGE // ====================================================== async function uploadPlayerImage( file, tournamentId, registrationId, playerNumber, side ) { if (!file) { return ""; } const extension = file.name .split(".") .pop() .toLowerCase(); const filePath = `registrations/${tournamentId}/${registrationId}/player-${playerNumber}/aadhaar-${side}.${extension}`; const storageRef = ref( storage, filePath ); await uploadBytes( storageRef, file, { contentType: file.type } ); const downloadURL = await getDownloadURL( storageRef ); return downloadURL; } // ====================================================== // TEAM FORM SUBMIT // ====================================================== form?.addEventListener( "submit", async (event) => { event.preventDefault(); if (!currentUser) { alert( "Please login first." ); return; } const submitBtn = form.querySelector( 'button[type="submit"]' ); try { if (submitBtn) { submitBtn.disabled = true; submitBtn.textContent = "Submitting..."; } // ---------------------------------------------- // TEAM DETAILS // ---------------------------------------------- const teamName = document.getElementById( "teamName" )?.value.trim() || ""; const captainName = document.getElementById( "captainName" )?.value.trim() || ""; const teamContact = document.getElementById( "teamContact" )?.value.trim() || ""; if (!teamName) { throw new Error( "Team name is required." ); } if (!captainName) { throw new Error( "Captain name is required." ); } if ( !/^[0-9]{10}$/.test( teamContact ) ) { throw new Error( "Enter valid 10 digit team contact number." ); } // ---------------------------------------------- // PLAYERS // ---------------------------------------------- const players = getPlayers(); // ---------------------------------------------- // CREATE REGISTRATION ID // ---------------------------------------------- const registrationRef = doc( collection( db, "registrations" ) ); const registrationId = registrationRef.id; // ---------------------------------------------- // UPLOAD PLAYER DOCUMENTS // ---------------------------------------------- const finalPlayers = []; for ( let index = 0; index < players.length; index++ ) { const player = players[index]; if (submitBtn) { submitBtn.textContent = `Uploading Player ${index + 1}/${players.length}...`; } const frontUrl = await uploadPlayerImage( player.frontFile, tournamentId, registrationId, player.index, "front" ); const backUrl = await uploadPlayerImage( player.backFile, tournamentId, registrationId, player.index, "back" ); finalPlayers.push({ playerNumber: player.index, name: player.name, mobile: player.mobile, mobileNumber: player.mobile, phone: player.mobile, dob: player.dob, dateOfBirth: player.dob, age: player.age, role: player.role, aadhaarNumber: player.aadhaarNumber, aadhaar: player.aadhaarNumber, aadhaarFrontUrl: frontUrl, aadhaarBackUrl: backUrl, aadhaarFront: frontUrl, aadhaarBack: backUrl }); } // ---------------------------------------------- // PAYMENT // ---------------------------------------------- const utr = document.getElementById( "utr" )?.value.trim() || ""; const paymentScreenshot = document.getElementById( "paymentScreenshot" )?.files?.[0] || null; // ---------------------------------------------- // PAYMENT SCREENSHOT // ---------------------------------------------- let paymentScreenshotUrl = ""; if (paymentScreenshot) { if (submitBtn) { submitBtn.textContent = "Uploading payment proof..."; } paymentScreenshotUrl = await uploadPlayerImage( paymentScreenshot, tournamentId, registrationId, "team", "payment" ); } // ---------------------------------------------- // FIRESTORE DATA // ---------------------------------------------- const registrationData = { // ID registrationId, tournamentId, // USER userId: currentUser.uid, uid: currentUser.uid, email: currentUser.email || "", googleName: currentUser.displayName || "", googlePhoto: currentUser.photoURL || "", // TEAM teamName, captainName, teamContact, contactNumber: teamContact, // TOURNAMENT tournamentName: getField( tournamentData, [ "name", "tournamentName", "title" ], "" ), sport: getField( tournamentData, [ "sport", "category", "type" ], "" ), // PLAYERS players: finalPlayers, playerCount: finalPlayers.length, minimumPlayers: minPlayers, maximumPlayers: maxPlayers, // PAYMENT utr, paymentScreenshotUrl, paymentStatus: "pending", // REGISTRATION status: "pending", approved: false, createdAt: serverTimestamp(), updatedAt: serverTimestamp() }; // ---------------------------------------------- // SAVE // ---------------------------------------------- if (submitBtn) { submitBtn.textContent = "Saving registration..."; } await addDoc( collection( db, "registrations" ), registrationData ); alert( "Registration submitted successfully!" ); window.location.href = "index.html"; } catch (error) { console.error( "REGISTRATION ERROR:", error ); alert( error.message || "Registration failed." ); if (submitBtn) { submitBtn.disabled = false; submitBtn.textContent = "Submit Registration"; } } } ); // ====================================================== // MOBILE INPUT ONLY NUMBERS // ====================================================== document.addEventListener( "input", (event) => { if ( event.target.matches( ".player-mobile" ) || event.target.matches( ".player-aadhaar" ) ) { event.target.value = event.target.value .replace( /\D/g, "" ); } } ); // ====================================================== // CONSOLE // ====================================================== console.log( "KPL Registration System Loaded" );
